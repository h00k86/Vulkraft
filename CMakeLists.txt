cmake_minimum_required(VERSION 3.20)
# Read the same public header used by Make and the source packager.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/ui/version.h")
foreach(component MAJOR MINOR PATCH)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/include/ui/version.h" version_line
       REGEX "^#define UI_VERSION_${component} (0|[1-9][0-9]*)$")
  list(LENGTH version_line version_count)
  if(NOT version_count EQUAL 1)
    message(FATAL_ERROR "Expected one decimal UI_VERSION_${component} in include/ui/version.h")
  endif()
  string(REGEX REPLACE "^#define UI_VERSION_${component} " "" UI_VERSION_${component} "${version_line}")
endforeach()
project(UIFramework VERSION "${UI_VERSION_MAJOR}.${UI_VERSION_MINOR}.${UI_VERSION_PATCH}" LANGUAGES C)
